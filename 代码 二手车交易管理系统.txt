CREATE SCHEMA `二手车交易管理系统` ;

CREATE TABLE `二手车交易管理系统`.`t_car` (
  `CarID` INT NOT NULL AUTO_INCREMENT COMMENT '车辆编号',
  `Brand` VARCHAR(20) NOT NULL COMMENT '车辆品牌',
  `Model` VARCHAR(50) NOT NULL COMMENT '车辆型号',
  `Year` INT NOT NULL COMMENT '年款',
  `Mileage` INT NOT NULL COMMENT '里程数（公里）',
  `RegisterDate` DATE NOT NULL COMMENT '初次登记日期',
  `Condition` VARCHAR(10) NOT NULL COMMENT '车况',
  `PurchasePrice` DECIMAL(10,2) NOT NULL COMMENT '收购价格（元）',
  `SellingPrice` DECIMAL(10,2) NOT NULL COMMENT '销售价格（元）',
  `VIN` VARCHAR(17) NOT NULL COMMENT '车架号',
  PRIMARY KEY (`CarID`),
  UNIQUE INDEX `VIN_UNIQUE` (`VIN` ASC) VISIBLE)
COMMENT = '车辆信息表';

CREATE TABLE `二手车交易管理系统`.`t_seller` (
  `SellerID` INT NOT NULL AUTO_INCREMENT COMMENT '卖家编号',
  `Name` VARCHAR(20) NOT NULL COMMENT '卖家姓名',
  `Phone` VARCHAR(11) NOT NULL COMMENT '联系电话',
  `City` VARCHAR(20) NOT NULL COMMENT '所在城市',
  `SellingModel` VARCHAR(50) NOT NULL COMMENT '待售车型',
  `ExpectedPrice` DECIMAL(10,2) NOT NULL COMMENT '期望售价（元）',
  `TransferWillingness` VARCHAR(5) NOT NULL COMMENT '过户意愿',
  `IDCard` VARCHAR(18) NOT NULL COMMENT '身份证号',
  PRIMARY KEY (`SellerID`),
  UNIQUE INDEX `Phone_UNIQUE` (`Phone` ASC) VISIBLE,
  UNIQUE INDEX `IDCard_UNIQUE` (`IDCard` ASC) VISIBLE)
COMMENT = '卖家信息表';

CREATE TABLE `二手车交易管理系统`.`t_buyer` (
  `BuyerID` INT NOT NULL AUTO_INCREMENT COMMENT '买家编号',
  `Name` VARCHAR(20) NOT NULL COMMENT '买家姓名',
  `Phone` VARCHAR(11) NOT NULL COMMENT '联系电话',
  `City` VARCHAR(20) NOT NULL COMMENT '所在城市',
  `IntendedModel` VARCHAR(50) NOT NULL COMMENT '意向车型',
  `Budget` VARCHAR(20) NOT NULL COMMENT '预算范围（元）',
  `IDCard` VARCHAR(18) NOT NULL COMMENT '身份证号',
  PRIMARY KEY (`BuyerID`),
  UNIQUE INDEX `Phone_UNIQUE` (`Phone` ASC) VISIBLE,
  UNIQUE INDEX `IDCard_UNIQUE` (`IDCard` ASC) VISIBLE)
COMMENT = '买家信息表';

CREATE TABLE `二手车交易管理系统`.`t_transaction` (
  `TransID` INT NOT NULL AUTO_INCREMENT COMMENT '交易编号',
  `DealPurchasePrice` DECIMAL(10,2) NOT NULL COMMENT '实际收购价格（元）',
  `DealSellingPrice` DECIMAL(10,2) NOT NULL COMMENT '实际销售价格（元）',
  `TransDate` DATE NOT NULL COMMENT '交易日期',
  `TransStatus` VARCHAR(10) NOT NULL COMMENT '交易状态',
  `CarID` INT NOT NULL COMMENT '关联车辆编号',
  `SellerID` INT NOT NULL COMMENT '关联卖家编号',
  `Buyer` INT NOT NULL COMMENT '关联买家编号',
  PRIMARY KEY (`TransID`),
  INDEX `CarID_idx` (`CarID` ASC) VISIBLE,
  INDEX `SellerID_idx` (`SellerID` ASC) VISIBLE,
  INDEX `BuyerID_idx` (`Buyer` ASC) VISIBLE,
  CONSTRAINT `CarID`
    FOREIGN KEY (`CarID`)
    REFERENCES `二手车交易管理系统`.`t_car` (`CarID`)
    ON DELETE RESTRICT
    ON UPDATE CASCADE,
  CONSTRAINT `SellerID`
    FOREIGN KEY (`SellerID`)
    REFERENCES `二手车交易管理系统`.`t_seller` (`SellerID`)
    ON DELETE RESTRICT
    ON UPDATE CASCADE,
  CONSTRAINT `BuyerID`
    FOREIGN KEY (`Buyer`)
    REFERENCES `二手车交易管理系统`.`t_buyer` (`BuyerID`)
    ON DELETE RESTRICT
    ON UPDATE CASCADE)
COMMENT = '交易信息表';

CREATE TABLE `二手车交易管理系统`.`t_maintenance_record` (
  `RecordID` INT NOT NULL AUTO_INCREMENT COMMENT '维修记录编号',
  `MaintenanceDate` DATE NOT NULL COMMENT '维修日期',
  `MaintenanceItem` VARCHAR(100) NOT NULL COMMENT '维修项目',
  `MaintenanceCost` DECIMAL(8,2) NOT NULL COMMENT '维修费用（元）',
  `MaintenanceOrg` VARCHAR(50) NOT NULL COMMENT '维修机构',
  `CarID` INT NOT NULL COMMENT '关联车辆编号',
  PRIMARY KEY (`RecordID`),
  INDEX `CarID_idx` (`CarID` ASC) VISIBLE,
  CONSTRAINT `CarIDD`
    FOREIGN KEY (`CarID`)
    REFERENCES `二手车交易管理系统`.`t_car` (`CarID`)
    ON DELETE CASCADE
    ON UPDATE CASCADE)
COMMENT = '车辆维修记录表';
