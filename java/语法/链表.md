# Java 链表系统学习指南

链表是一种重要的数据结构，我将从基础到高级为你系统讲解，并提供完整的Java实现。

## 1. 链表基础概念

### 什么是链表？
链表是一种线性数据结构，通过节点间的指针连接实现。与数组不同，链表在内存中不是连续存储的。

### 链表 vs 数组
- **数组**：连续内存，随机访问快，插入删除慢
- **链表**：非连续内存，顺序访问，插入删除快

```java
public class ListNode {
    int val;        // 节点值
    ListNode next;  // 指向下一个节点的指针
    
    // 构造函数
    public ListNode(int val) {
        this.val = val;
        this.next = null;
    }
    
    public ListNode(int val, ListNode next) {
        this.val = val;
        this.next = next;
    }
}
```

## 3. 单链表实现

```java
public class SinglyLinkedList {
    private ListNode head;  // 头节点
    private int size;       // 链表长度
    
    public SinglyLinkedList() {
        this.head = null;
        this.size = 0;
    }
    
    // 在链表头部添加节点
    public void addFirst(int val) {
        ListNode newNode = new ListNode(val);
        newNode.next = head;
        head = newNode;
        size++;
    }
    
    // 在链表尾部添加节点
    public void addLast(int val) {
        ListNode newNode = new ListNode(val);
        
        if (head == null) {
            head = newNode;
        } else {
            ListNode current = head;
            while (current.next != null) {
                current = current.next;
            }
            current.next = newNode;
        }
        size++;
    }
    
    // 在指定位置插入节点
    public void addAtIndex(int index, int val) {
        if (index < 0 || index > size) {
            throw new IndexOutOfBoundsException("Index: " + index + ", Size: " + size);
        }
        
        if (index == 0) {
            addFirst(val);
            return;
        }
        
        ListNode newNode = new ListNode(val);
        ListNode prev = head;
        
        // 找到要插入位置的前一个节点
        for (int i = 0; i < index - 1; i++) {
            prev = prev.next;
        }
        
        newNode.next = prev.next;
        prev.next = newNode;
        size++;
    }
    
    // 删除第一个节点
    public void deleteFirst() {
        if (head == null) return;
        
        head = head.next;
        size--;
    }
    
    // 删除最后一个节点
    public void deleteLast() {
        if (head == null) return;
        
        if (head.next == null) {
            head = null;
        } else {
            ListNode current = head;
            while (current.next.next != null) {
                current = current.next;
            }
            current.next = null;
        }
        size--;
    }
    
    // 删除指定位置的节点
    public void deleteAtIndex(int index) {
        if (index < 0 || index >= size) {
            throw new IndexOutOfBoundsException("Index: " + index + ", Size: " + size);
        }
        
        if (index == 0) {
            deleteFirst();
            return;
        }
        
        ListNode prev = head;
        for (int i = 0; i < index - 1; i++) {
            prev = prev.next;
        }
        
        prev.next = prev.next.next;
        size--;
    }
    
    // 查找节点
    public boolean contains(int val) {
        ListNode current = head;
        while (current != null) {
            if (current.val == val) {
                return true;
            }
            current = current.next;
        }
        return false;
    }
    
    // 获取节点值
    public int get(int index) {
        if (index < 0 || index >= size) {
            throw new IndexOutOfBoundsException("Index: " + index + ", Size: " + size);
        }
        
        ListNode current = head;
        for (int i = 0; i < index; i++) {
            current = current.next;
        }
        return current.val;
    }
    
    // 获取链表长度
    public int size() {
        return size;
    }
    
    // 判断链表是否为空
    public boolean isEmpty() {
        return size == 0;
    }
    
    // 打印链表
    public void display() {
        ListNode current = head;
        System.out.print("链表: ");
        while (current != null) {
            System.out.print(current.val + " -> ");
            current = current.next;
        }
        System.out.println("null");
    }
}
```

## 4. 双向链表实现

```java
// 双向链表节点
class DoublyListNode {
    int val;
    DoublyListNode prev;
    DoublyListNode next;
    
    public DoublyListNode(int val) {
        this.val = val;
        this.prev = null;
        this.next = null;
    }
}

public class DoublyLinkedList {
    private DoublyListNode head;
    private DoublyListNode tail;
    private int size;
    
    public DoublyLinkedList() {
        this.head = null;
        this.tail = null;
        this.size = 0;
    }
    
    // 在头部添加节点
    public void addFirst(int val) {
        DoublyListNode newNode = new DoublyListNode(val);
        
        if (head == null) {
            head = tail = newNode;
        } else {
            newNode.next = head;
            head.prev = newNode;
            head = newNode;
        }
        size++;
    }
    
    // 在尾部添加节点
    public void addLast(int val) {
        DoublyListNode newNode = new DoublyListNode(val);
        
        if (tail == null) {
            head = tail = newNode;
        } else {
            tail.next = newNode;
            newNode.prev = tail;
            tail = newNode;
        }
        size++;
    }
    
    // 删除头部节点
    public void deleteFirst() {
        if (head == null) return;
        
        if (head == tail) {
            head = tail = null;
        } else {
            head = head.next;
            head.prev = null;
        }
        size--;
    }
    
    // 删除尾部节点
    public void deleteLast() {
        if (tail == null) return;
        
        if (head == tail) {
            head = tail = null;
        } else {
            tail = tail.prev;
            tail.next = null;
        }
        size--;
    }
    
    // 正向打印
    public void displayForward() {
        DoublyListNode current = head;
        System.out.print("正向: ");
        while (current != null) {
            System.out.print(current.val + " <-> ");
            current = current.next;
        }
        System.out.println("null");
    }
    
    // 反向打印
    public void displayBackward() {
        DoublyListNode current = tail;
        System.out.print("反向: ");
        while (current != null) {
            System.out.print(current.val + " <-> ");
            current = current.prev;
        }
        System.out.println("null");
    }
}
```

## 5. 循环链表实现

```java
public class CircularLinkedList {
    private ListNode tail;  // 使用尾指针更方便
    private int size;
    
    public CircularLinkedList() {
        this.tail = null;
        this.size = 0;
    }
    
    // 在尾部添加节点
    public void add(int val) {
        ListNode newNode = new ListNode(val);
        
        if (tail == null) {
            tail = newNode;
            tail.next = tail;  // 指向自己形成环
        } else {
            newNode.next = tail.next;
            tail.next = newNode;
            tail = newNode;
        }
        size++;
    }
    
    // 删除节点
    public void delete(int val) {
        if (tail == null) return;
        
        // 只有一个节点的情况
        if (tail == tail.next && tail.val == val) {
            tail = null;
            size--;
            return;
        }
        
        ListNode current = tail;
        do {
            ListNode nextNode = current.next;
            if (nextNode.val == val) {
                current.next = nextNode.next;
                if (nextNode == tail) {
                    tail = current;
                }
                size--;
                return;
            }
            current = current.next;
        } while (current != tail);
    }
    
    // 打印循环链表
    public void display() {
        if (tail == null) {
            System.out.println("链表为空");
            return;
        }
        
        ListNode current = tail.next;
        System.out.print("循环链表: ");
        do {
            System.out.print(current.val + " -> ");
            current = current.next;
        } while (current != tail.next);
        System.out.println("(回到起点)");
    }
}
```

## 6. 常见算法练习

### 6.1 反转链表
```java
public class LinkedListAlgorithms {
    
    // 迭代法反转链表
    public static ListNode reverseList(ListNode head) {
        ListNode prev = null;
        ListNode current = head;
        
        while (current != null) {
            ListNode nextTemp = current.next;
            current.next = prev;
            prev = current;
            current = nextTemp;
        }
        
        return prev;
    }
    
    // 递归法反转链表
    public static ListNode reverseListRecursive(ListNode head) {
        if (head == null || head.next == null) {
            return head;
        }
        
        ListNode newHead = reverseListRecursive(head.next);
        head.next.next = head;
        head.next = null;
        
        return newHead;
    }
}
```

### 6.2 检测环
```java
// 检测链表是否有环（快慢指针法）
public static boolean hasCycle(ListNode head) {
    if (head == null || head.next == null) {
        return false;
    }
    
    ListNode slow = head;
    ListNode fast = head;
    
    while (fast != null && fast.next != null) {
        slow = slow.next;
        fast = fast.next.next;
        
        if (slow == fast) {
            return true;
        }
    }
    
    return false;
}
```

```java
// 找到链表的中间节点
public static ListNode findMiddle(ListNode head) {
    if (head == null) return null;
    
    ListNode slow = head;
    ListNode fast = head;
    
    while (fast != null && fast.next != null) {
        slow = slow.next;
        fast = fast.next.next;
    }
    
    return slow;
}
```

### 6.4 合并两个有序链表
```java
// 合并两个有序链表
public static ListNode mergeTwoLists(ListNode l1, ListNode l2) {
    ListNode dummy = new ListNode(0);
    ListNode current = dummy;
    
    while (l1 != null && l2 != null) {
        if (l1.val <= l2.val) {
            current.next = l1;
            l1 = l1.next;
        } else {
            current.next = l2;
            l2 = l2.next;
        }
        current = current.next;
    }
    
    // 连接剩余部分
    if (l1 != null) {
        current.next = l1;
    } else {
        current.next = l2;
    }
    
    return dummy.next;
}
```
