[柠檬水找零](https://leetcode.cn/problems/lemonade-change/)
丢这题上来只是警醒一下自己不要什么都怕，好歹先写一下再说。
```java
class Solution {
    public boolean lemonadeChange(int[] bills) {
        int five=0,ten=0;
        for(int bill:bills){
            if(bill==5){
                five++;
            }
            else if(bill==10){
                if(five!=0){
                    ten++;five--;
                }
                else return false;
            }
            else if(bill==20){
                if(ten!=0&&five!=0)
                {
                    ten--;five--;
                }
                else if(ten==0&&five>=3){
                    five-=3;
                }
                else return false;
            }
        }
        return true;
    }
}
```
